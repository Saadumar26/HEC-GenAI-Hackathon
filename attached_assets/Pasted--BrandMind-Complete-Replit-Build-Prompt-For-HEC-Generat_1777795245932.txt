## PROJECT OVERVIEW

**BrandMind** is an AI-powered agentic system that:
1. Analyzes a company's brand identity from their website and existing social posts
2. Takes a simple content intent from the user
3. Automatically generates on-brand LinkedIn and Instagram post variations (caption + image)
4. Runs a self-review loop (3 rounds) to score and improve each post
5. Presents results to the human for final selection and editing

---

## TECH STACK — STRICT REQUIREMENT

**Frontend:** React.js (Vite) + Tailwind CSS
**Backend:** Python FastAPI
**AI/LLM:** OpenAI GPT-4o (captions, brand analysis, self-review)
**Image Generation:** OpenAI DALL-E 3
**Web Scraping:** Firecrawl API (or BeautifulSoup as fallback)
**Agent Orchestration:** LangChain (or direct OpenAI API calls)
**HTTP Client:** Axios (frontend) + httpx (backend)
**CORS:** FastAPI CORS middleware
**Environment:** .env file for all API keys

```python
import os
import json
import requests
from bs4 import BeautifulSoup
from openai import OpenAI

client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

def scrape_website(url: str) -> str:
    """Scrape website content using BeautifulSoup"""
    try:
        headers = {"User-Agent": "Mozilla/5.0"}
        response = requests.get(url, headers=headers, timeout=10)
        soup = BeautifulSoup(response.content, "html.parser")
        
        # Extract meaningful text
        for tag in soup(["script", "style", "nav", "footer"]):
            tag.decompose()
        
        text = soup.get_text(separator=" ", strip=True)
        return text[:3000]  # Limit to 3000 chars
    except Exception as e:
        return f"Could not scrape website: {str(e)}"

def analyze_brand(website_url: str, brand_name: str) -> dict:
    """Analyze brand from website and return brand profile"""
    
    website_content = scrape_website(website_url)
    
    prompt = f"""
    You are a brand strategist. Analyze this website content and extract the brand profile.
    
    Brand Name: {brand_name}
    Website Content: {website_content}
    
    Return ONLY a valid JSON object with these exact fields:
    {{
        "brand_name": "{brand_name}",
        "tone": "one of: professional, casual, energetic, inspirational, technical",
        "emoji_usage": "one of: none, minimal, moderate, heavy",
        "avg_caption_length_linkedin": 150,
        "avg_caption_length_instagram": 80,
        "color_palette": ["#hex1", "#hex2", "#hex3"],
        "cta_style": "one of: action_oriented, soft, question_based",
        "hashtag_count": 5,
        "visual_style": "one of: minimalist, bold_modern, corporate, playful, technical",
        "keywords": ["keyword1", "keyword2", "keyword3", "keyword4", "keyword5"],
        "voice_description": "2 sentence description of brand voice"
    }}
    
    Return ONLY JSON, no explanation, no markdown.
    """
    
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        response_format={"type": "json_object"}
    )
    
    return json.loads(response.choices[0].message.content)
```

```python
import os
import json
from openai import OpenAI

client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

def generate_captions(
    platform: str,
    content_brief: dict,
    brand_profile: dict,
    num_variations: int = 3
) -> list:
    """Generate caption variations for a platform"""
    
    max_length = (
        brand_profile.get("avg_caption_length_linkedin", 150)
        if platform == "linkedin"
        else brand_profile.get("avg_caption_length_instagram", 80)
    )
    
    prompt = f"""
    You are a professional social media copywriter.
    
    Platform: {platform}
    Brand Profile: {json.dumps(brand_profile)}
    Content Brief: {json.dumps(content_brief)}
    Max caption length: {max_length} words
    Emoji usage: {brand_profile.get('emoji_usage', 'moderate')}
    
    Generate {num_variations} DIFFERENT caption variations.
    Each variation must have a completely different hook and angle.
    
    Return ONLY a valid JSON object:
    {{
        "variations": [
            {{
                "variation_number": 1,
                "caption": "full caption text here",
                "overlay_text": "short 4-6 word overlay for image",
                "hook_type": "question/statement/stat/story"
            }},
            {{
                "variation_number": 2,
                "caption": "full caption text here",
                "overlay_text": "short 4-6 word overlay for image",
                "hook_type": "question/statement/stat/story"
            }},
            {{
                "variation_number": 3,
                "caption": "full caption text here",
                "overlay_text": "short 4-6 word overlay for image",
                "hook_type": "question/statement/stat/story"
            }}
        ]
    }}
    
    Return ONLY JSON.
    """
    
    response = client.chat.completions.create(
        model="gpt-4o",
        messages=[{"role": "user", "content": prompt}],
        response_format={"type": "json_object"}
    )
    
    result = json.loads(response.choices[0].message.content)
    return result.get("variations", [])


def generate_image(
    brand_profile: dict,
    content_brief: dict,
    platform: str
) -> str:
    """Generate image using DALL-E 3"""
    
    size = "1792x1024" if platform == "linkedin" else "1024x1024"
    
    colors = brand_profile.get("color_palette", ["#1A1A2E", "#FF6B35"])
    visual_style = brand_profile.get("visual_style", "bold_modern")
    visual_direction = content_brief.get("visual_direction", "professional modern design")
    brand_name = brand_profile.get("brand_name", "Brand")
    
    image_prompt = f"""
    Create a professional social media background graphic for {brand_name}.
    Style: {visual_style}, modern, clean.
    Theme: {visual_direction}.
    Color scheme: Use {colors[0]} as primary and {colors[1] if len(colors) > 1 else '#ffffff'} as accent.
    Important: NO text in the image. Clean background suitable for text overlay.
    Platform: {platform}. High quality, professional look.
    """
    
    try:
        response = client.images.generate(
            model="dall-e-3",
            prompt=image_prompt,
            size=size,
            quality="standard",
            n=1
        )
        return response.data[0].url
    except Exception as e:
        # Return placeholder if generation fails
        return f"https://placehold.co/{size.replace('x', '/')}?text={brand_name}"
```

### File: backend/agents/self_review.py

```python
import os
import json
from openai import OpenAI

client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

def review_post(
    caption: str,
    platform: str,
    brand_profile: dict,
    content_brief: dict
) -> dict:
    """Run 3-round self-review on a post"""
    
    scores = []
    current_caption = caption
    
    rounds = [
        {
            "name": "Brand consistency",
            "criteria": "Does the tone, voice, emoji usage, and CTA style match the brand profile exactly?"
        },
        {
            "name": "Message clarity", 
            "criteria": "Is the main message clear? Is the hook engaging? Is the CTA actionable?"
        },
        {
            "name": "Platform optimization",
            "criteria": f"Is this optimized for {platform}? Correct length, format, hashtags?"
        }
    ]
    
    for round_info in rounds:
        prompt = f"""
        You are a senior social media quality reviewer.
        
        Round: {round_info['name']}
        Criteria: {round_info['criteria']}
        Platform: {platform}
        Brand Profile: {json.dumps(brand_profile)}
        
        Caption to review:
        {current_caption}
        
        Return ONLY a valid JSON object:
        {{
            "round_name": "{round_info['name']}",
            "score": 8.5,
            "passed": true,
            "main_issue": "one sentence issue if any",
            "improved_caption": "improved version of caption (keep original if score >= 8)"
        }}
        
        Score must be between 0-10. Return ONLY JSON.
        """
        
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        scores.append(result)
        
        # Update caption if improved
        if result.get("improved_caption"):
            current_caption = result["improved_caption"]
    
    # Calculate final score
    avg_score = sum(r.get("score", 7) for r in scores) / len(scores)
    
    return {
        "final_caption": current_caption,
        "final_score": round(avg_score, 1),
        "rounds": scores,
        "passed": avg_score >= 7.0
    }
```

### File: backend/agents/orchestrator.py

```python
import asyncio
from agents.brand_analyzer import analyze_brand
from agents.content_planner import create_content_brief
from agents.creative_generator import generate_captions, generate_image
from agents.self_review import review_post

def run_full_pipeline(
    website_url: str,
    brand_name: str,
    intent: str,
    platforms: list,
    tone_override: str = None
) -> dict:
    """Main orchestrator — runs all agents in sequence"""
    
    # Step 1: Brand Analysis
    print("Step 1: Analyzing brand...")
    brand_profile = analyze_brand(website_url, brand_name)
    
    # Step 2: Content Planning
    print("Step 2: Planning content...")
    content_brief = create_content_brief(
        intent, platforms, tone_override, brand_profile
    )
    
    # Step 3 + 4: Generate + Review for each platform
    results = {}
    
    for platform in platforms:
        print(f"Step 3: Generating captions for {platform}...")
        captions = generate_captions(platform, content_brief, brand_profile)
        
        print(f"Step 3b: Generating image for {platform}...")
        image_url = generate_image(brand_profile, content_brief, platform)
        
        platform_posts = []
        best_score = 0
        best_index = 0
        
        for i, cap_data in enumerate(captions):
            print(f"Step 4: Reviewing {platform} variation {i+1}...")
            
            review_result = review_post(
                cap_data["caption"],
                platform,
                brand_profile,
                content_brief
            )
            
            post = {
                "variation_number": i + 1,
                "caption": review_result["final_caption"],
                "image_url": image_url,
                "overlay_text": cap_data.get("overlay_text", ""),
                "review_score": review_result["final_score"],
                "review_notes": review_result["rounds"][-1].get("main_issue", "Looks great!"),
                "platform": platform,
                "recommended": False
            }
            
            platform_posts.append(post)
            
            if review_result["final_score"] > best_score:
                best_score = review_result["final_score"]
                best_index = i
        
        # Mark best as recommended
        if platform_posts:
            platform_posts[best_index]["recommended"] = True
        
        results[platform] = platform_posts
    
    return {
        "brand_profile": brand_profile,
        "content_brief": content_brief,
        "posts": results
    }
```

```python
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import List, Optional
from agents.orchestrator import run_full_pipeline
from agents.creative_generator import generate_captions
from agents.self_review import review_post

router = APIRouter()

class GenerateRequest(BaseModel):
    website_url: str
    brand_name: str
    intent: str
    platforms: List[str]
    tone_override: Optional[str] = None

class EditRequest(BaseModel):
    caption: str
    edit_instruction: str
    platform: str
    brand_profile: dict
    content_brief: dict

@router.post("/generate")
def generate_posts(request: GenerateRequest):
    """Main endpoint — runs full pipeline"""
    try:
        result = run_full_pipeline(
            website_url=request.website_url,
            brand_name=request.brand_name,
            intent=request.intent,
            platforms=request.platforms,
            tone_override=request.tone_override
        )
        return {"success": True, "data": result}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/edit")
def edit_post(request: EditRequest):
    """Edit a single post variation"""
    try:
        from openai import OpenAI
        import os
        
        client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
        
        prompt = f"""
        Edit this {request.platform} caption based on the instruction.
        
        Original caption: {request.caption}
        Edit instruction: {request.edit_instruction}
        Brand profile tone: {request.brand_profile.get('tone', 'professional')}
        
        Return the edited caption only. No explanation.
        """
        
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}]
        )
        
        edited_caption = response.choices[0].message.content.strip()
        
        # Re-review edited caption
        review = review_post(
            edited_caption,
            request.platform,
            request.brand_profile,
            request.content_brief
        )
        
        return {
            "success": True,
            "caption": review["final_caption"],
            "score": review["final_score"]
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/health")
def health_check():
    return {"status": "healthy", "service": "BrandMind API"}
```

```jsx
import { useState, useEffect } from 'react'
import { generatePosts } from '../api/brandmind'

const STEPS_LOG = [
  'Analyzing brand identity from website...',
  'Extracting tone, voice and visual style...',
  'Creating structured content brief...',
  'Generating caption variations...',
  'Creating post images with DALL-E 3...',
  'Running self-review round 1 — brand consistency...',

  'Running self-review round 3 — platform optimization...',

  'Finalizing posts...'

export default function Step3_Generate({ brandData, contentPlan, onGenerated, onBack }) {
  const [loading, setLoading] = useState(false)
  const [logIndex, setLogIndex] = useState(0)
  const [error, setError] = useState(null)

  const handleGenerate = async () => {
    setLoading(true)
    setError(null)
    setLogIndex(0)

    // Simulate progress logs
    const interval = setInterval(() => {
      setLogIndex(prev => {
        if (prev < STEPS_LOG.length - 1) return prev + 1
        clearInterval(interval)
        return prev
      })
    }, 2500)

    try {
      const result = await generatePosts({
        website_url: brandData.websiteUrl,
        brand_name: brandData.brandName,
        intent: contentPlan.intent,
        platforms: contentPlan.platforms,
        tone_override: contentPlan.toneOverride || null
      })

      clearInterval(interval)
      onGenerated(result.data)
    } catch (err) {
      clearInterval(interval)
      setError(err.response?.data?.detail || 'Something went wrong. Please try again.')
      setLoading(false)
    }
  }

  return (
    <div className="max-w-lg mx-auto">
      <h1 className="text-2xl font-medium text-gray-900 mb-1">Generate posts</h1>
```

```jsx
import { useState } from 'react'
import { editPost } from '../api/brandmind'
import PostCard from '../components/PostCard'

export default function Step4_Review({ generatedPosts, onBack }) {
  const [activePlatform, setActivePlatform] = useState(
    Object.keys(generatedPosts.posts)[0]
  )
  const [posts, setPosts] = useState(generatedPosts.posts)
  const [editingPost, setEditingPost] = useState(null)
  const [editInstruction, setEditInstruction] = useState('')
  const [editLoading, setEditLoading] = useState(false)
  const [selectedPosts, setSelectedPosts] = useState([])

  const handleEdit = async (post) => {
    if (!editInstruction.trim()) return
    setEditLoading(true)

    try {
      const result = await editPost({
        caption: post.caption,
        edit_instruction: editInstruction,
        platform: post.platform,
        brand_profile: generatedPosts.brand_profile,
        content_brief: generatedPosts.content_brief
      })

      setPosts(prev => ({
        ...prev,
        [activePlatform]: prev[activePlatform].map(p =>
          p.variation_number === post.variation_number
            ? { ...p, caption: result.caption, review_score: result.score }
            : p
        )
      }))

      setEditingPost(null)
      setEditInstruction('')
    } catch (e) {
      alert('Edit failed. Please try again.')
    } finally {
      setEditLoading(false)
    }
  }

  const handleSelect = (post) => {
    setSelectedPosts(prev => {
      const exists = prev.find(p => p.variation_number === post.variation_number && p.platform === post.platform)
      if (exists) return prev.filter(p => !(p.variation_number === post.variation_number && p.platform === post.platform))
      return [...prev, post]
    })
  }

  const handleCopy = (caption) => {
    navigator.clipboard.writeText(caption)
    alert('Caption copied to clipboard!')
  }

  return (
    <div>
      <div className="flex items-center justify-between mb-6">
        <div>
          <h1 className="text-2xl font-medium text-gray-900">Review & export</h1>
          <p className="text-sm text-gray-500">Select your preferred variations, edit if needed, then copy or download.</p>
        </div>
        {selectedPosts.length > 0 && (
          <div className="bg-green-50 border border-green-200 rounded-lg px-4 py-2">
            <p className="text-sm text-green-700 font-medium">{selectedPosts.length} post{selectedPosts.length > 1 ? 's' : ''} selected</p>
          </div>
        )}
      </div>

      {/* Brand Profile Summary */}
      <div className="bg-white border border-gray-100 rounded-xl p-4 mb-6 flex items-center gap-4">
        <div className="flex-1">
          <p className="text-sm font-medium text-gray-900">{generatedPosts.brand_profile.brand_name}</p>
          <div className="flex flex-wrap gap-1 mt-1">
            {generatedPosts.brand_profile.keywords?.slice(0, 4).map(k => (
              <span key={k} className="text-xs px-2 py-0.5 bg-gray-100 text-gray-500 rounded-full">{k}</span>
            ))}
          </div>
        </div>
        <div className="flex gap-1">
          {generatedPosts.brand_profile.color_palette?.map((c, i) => (
            <div key={i} className="w-6 h-6 rounded border border-gray-200" style={{ backgroundColor: c }}></div>
          ))}
        </div>
      </div>

      {/* Platform Tabs */}
      <div className="flex gap-2 mb-4">
        {Object.keys(posts).map(platform => (
          <button
            key={platform}
            onClick={() => setActivePlatform(platform)}
            className={`px-4 py-2 rounded-lg text-sm font-medium capitalize transition-all ${
              activePlatform === platform
                ? 'bg-purple-500 text-white'
                : 'bg-white border border-gray-200 text-gray-600'
            }`}
          >
            {platform}
          </button>
        ))}
      </div>

      {/* Post Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {posts[activePlatform]?.map(post => (
          <PostCard
            key={post.variation_number}
            post={post}
            isSelected={selectedPosts.some(p => p.variation_number === post.variation_number && p.platform === post.platform)}
            isEditing={editingPost?.variation_number === post.variation_number}
            editInstruction={editInstruction}
            editLoading={editLoading}
            onSelect={() => handleSelect(post)}
            onEdit={() => setEditingPost(post)}
            onCopy={() => handleCopy(post.caption)}
            onEditSubmit={() => handleEdit(post)}
            onEditChange={setEditInstruction}
            onEditCancel={() => { setEditingPost(null); setEditInstruction('') }}
          />
        ))}
      </div>

      <div className="mt-6 flex gap-3">
        <button
          onClick={onBack}
          className="px-4 py-2.5 border border-gray-200 text-gray-600 rounded-lg text-sm"
        >
          ← Generate new
        </button>
      </div>
    </div>
  )
}
```
